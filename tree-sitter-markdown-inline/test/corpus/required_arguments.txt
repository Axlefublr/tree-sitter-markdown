================================================================================
Non-space-leading angle-bracketed text is a required argument
================================================================================
<string?>
--------------------------------------------------------------------------------
(inline
  (required_argument))

================================================================================
Valid HTML-looking tags are required arguments too
================================================================================
<bool>
--------------------------------------------------------------------------------
(inline
  (required_argument))

================================================================================
Closing HTML-looking tags are required arguments too
================================================================================
</bool>
--------------------------------------------------------------------------------
(inline
  (required_argument))

================================================================================
Angle-bracketed text with attributes is a required argument
================================================================================
<div class="note">
--------------------------------------------------------------------------------
(inline
  (required_argument))

================================================================================
URI autolinks also follow the required-argument rule
================================================================================
<https://example.com>
--------------------------------------------------------------------------------
(inline
  (required_argument))

================================================================================
HTML comments also follow the required-argument rule
================================================================================
<!-- note -->
--------------------------------------------------------------------------------
(inline
  (required_argument))
